    $LIST trong ObjectScript¶

    $LIST Là Gì?¶

    $LIST là kiểu dữ liệu danh sách có thứ tự native của IRIS — lưu nhiều giá trị trong một string duy nhất với binary encoding (không phải CSV).

    ⚠️ Không đọc trực tiếp bằng mắt được — phải dùng $LIST functions để thao tác.

    1. Tạo List¶

    ; $LISTBUILD (viết tắt: $LB)
SET colors = $LISTBUILD("Red", "Green", "Blue")
SET mixed  = $LISTBUILD("Alice", 30, "", "active")  ; có thể mix kiểu, có thể rỗng

; Tạo list rỗng
SET empty = $LISTBUILD()

    2. Truy Cập Phần Tử — $LIST / $LG¶

    SET fruits = $LB("Apple", "Banana", "Cherry")

; $LIST(list, index) — lấy phần tử tại index (bắt đầu từ 1)
WRITE $LIST(fruits, 1)   ; Apple
WRITE $LIST(fruits, 2)   ; Banana
WRITE $LIST(fruits, 3)   ; Cherry

; $LG là alias của $LIST (viết tắt: List Get)
WRITE $LG(fruits, 2)     ; Banana

; Index âm: đếm từ cuối
WRITE $LG(fruits, -1)    ; Cherry (phần tử cuối)

    3. Độ Dài List — $LISTLENGTH / $LL¶

    SET fruits = $LB("Apple", "Banana", "Cherry")
WRITE $LISTLENGTH(fruits)   ; 3
WRITE $LL(fruits)            ; 3 (viết tắt)

    4. Duyệt List¶

    SET fruits = $LB("Apple", "Banana", "Cherry")

; Cách 1: FOR loop với $LL
FOR i = 1:1:$LL(fruits) {
    WRITE i, ": ", $LG(fruits, i), !
}

; Cách 2: $LISTNEXT (hiệu quả hơn cho list lớn)
SET pointer = 0
WHILE $LISTNEXT(fruits, pointer, value) {
    WRITE value, !
}

    5. Tìm Kiếm — $LISTFIND / $LF¶

    SET fruits = $LB("Apple", "Banana", "Cherry")

; $LISTFIND trả về index nếu tìm thấy, 0 nếu không
WRITE $LISTFIND(fruits, "Banana")   ; 2
WRITE $LISTFIND(fruits, "Mango")    ; 0

IF $LISTFIND(fruits, "Apple") {
    WRITE "Found Apple!", !
}

    6. Chỉnh Sửa List¶

    SET list = $LB("a", "b", "c")

; Thêm vào cuối — dùng _ (concatenation)
SET list = list _ $LB("d")
WRITE $LL(list)   ; 4

; Thay thế phần tử tại vị trí n
SET list = $LIST(list, 1, 1, $LB("A"))  ; thay "a" thành "A"
WRITE $LG(list, 1)   ; A

; Lấy sublist (slice)
SET sub = $LIST(list, 2, 3)   ; lấy index 2 đến 3
WRITE $LL(sub)   ; 2

    7. Kiểm Tra Phần Tử Có Hợp Lệ Không — $LISTVALID¶

    ; Kiểm tra xem biến có phải là $LIST hợp lệ không
IF $LISTVALID(myVar) {
    WRITE "Đây là list hợp lệ"
}

    8. List vs Array — Khi Nào Dùng Cái Nào?¶

    $LIST Multidim Array
    Cấu trúc Ordered, indexed (1,2,3...) Có thể dùng bất kỳ key
    Lưu trong 1 biến ✅ (1 string) ❌ (nhiều nodes)
    Truyền vào method Dễ (pass by value) Cần ByRef
    Dùng trong Property class ✅ List Of %String ❌
    Performance với list lớn Chậm hơn Nhanh hơn
    Thường dùng khi Danh sách nhỏ, truyền qua lại Dataset lớn, cần lookup nhanh

    🖥️ Terminal Snippets¶

    ; Tạo và duyệt list
SET colors = $LB("Red","Green","Blue")
FOR i=1:1:$LL(colors) { WRITE i,": ",$LG(colors,i),! }

; $LISTNEXT
SET p=0  WHILE $LISTNEXT(colors,p,v) { WRITE v,! }

; Tìm kiếm
WRITE "$LISTFIND: ", $LISTFIND(colors,"Green"), !   ; 2

; Thêm phần tử
SET colors = colors _ $LB("Yellow")
WRITE "Length sau khi thêm: ", $LL(colors), !
