    ObjectScript Cheatsheet¶

    Variables & Control Flow¶

    SET x = 10
SET name = "Alice"
SET ^GlobalVar = "persistent"       ; global (persists)

IF x > 10 { WRITE "Greater" } ELSEIF x = 10 { WRITE "Equal" } ELSE { WRITE "Less" }
FOR i = 1:1:10 { WRITE i, ! }       ; from 1, step 1, to 10
WHILE x > 0 { SET x = x - 1 }
DO { SET x = x + 1 } WHILE x < 10

WRITE value, !                       ; ! = newline
KILL x                               ; delete variable
QUIT value                           ; return from method

    Built-in Functions¶

    Function Purpose
    $LENGTH(str) String length
    $LENGTH(str, ",") Number of delimited pieces
    $EXTRACT(str, from, to) Substring by position
    $EXTRACT(str, *) Last character
    $PIECE(str, delim, n) Get nth piece
    $PIECE(str, delim, n, m) Get pieces n through m
    $FIND(str, target) Position after target (not at)
    $REPLACE(str, old, new) String replacement
    $TRANSLATE(str, from, to) Character mapping
    $TRANSLATE(str, " ") Remove spaces
    $UPPERCASE(str) To uppercase
    $LOWERCASE(str) To lowercase
    $JUSTIFY(val, width) Right-pad to width
    $RANDOM(n) Random 0 to n-1
    $HOROLOG Internal date "days,seconds"
    $ZDATETIME($HOROLOG, 3) Format as YYYY-MM-DD HH:MM:SS

    Arrays (Local & Global)¶

    ; Assign
SET arr(1) = "Alice"
SET arr("key") = "string key"
SET person(1, "name") = "Alice"      ; multidimensional

; $ORDER traversal (standard pattern)
SET key = ""
FOR {
    SET key = $ORDER(arr(key))
    QUIT:key=""
    WRITE key, ": ", arr(key), !
}

; Reverse
SET key = $ORDER(arr(""), -1)
FOR {
    QUIT:key=""
    WRITE key, !
    SET key = $ORDER(arr(key), -1)
}

; $DATA — 0=not exist, 1=data, 10=children only, 11=both
IF $DATA(arr(key)) { WRITE arr(key) }

; Delete
KILL arr(1)                          ; delete node + subtree
KILL arr                             ; delete entire array

; Deep copy
MERGE dest = source

    Gotchas

    • $ORDER(arr("")) returns first key; returns "" when done
    • Pass arrays by reference: Method(ByRef pArr) → call as DO ..Method(.myArr)
    • KILL is permanent — no undo

    Lists ($LIST)¶

    SET colors = $LISTBUILD("Red", "Green", "Blue")   ; create
SET colors = $LB("Red", "Green", "Blue")           ; shorthand

WRITE $LIST(colors, 1)               ; Red  (1-indexed)
WRITE $LG(colors, -1)                ; last element
WRITE $LISTLENGTH(colors)            ; 3
WRITE $LL(colors)                    ; shorthand

; Efficient iteration
SET pointer = 0
WHILE $LISTNEXT(colors, pointer, value) {
    WRITE value, !
}

SET colors = colors _ $LB("Yellow") ; append
WRITE $LISTFIND(colors, "Green")    ; 2 (0 if not found)
IF $LISTVALID(myVar) { ... }        ; validate

    Gotchas

    • Lists use binary encoding — don't WRITE raw
    • No built-in "remove by value" — must reconstruct
    • $LISTNEXT is more efficient than FOR loop for large lists

    String Operations¶

    ; Concatenation
SET full = first _ " " _ last

; $PIECE — most common pattern for delimited data
SET csv = "Alice,30,Engineer"
WRITE $PIECE(csv, ",", 1)            ; Alice
WRITE $PIECE(csv, ",", 2, 3)         ; 30,Engineer
SET $PIECE(csv, ",", 2) = 31         ; replace in-place → "Alice,31,Engineer"

; $EXTRACT — by position
WRITE $EXTRACT(str, 1, 5)            ; chars 1–5
WRITE $EXTRACT(str, *)               ; last char
WRITE $EXTRACT(str, *-1, *)          ; last 2 chars
SET $EXTRACT(str, 1, 5) = "Hyper"    ; replace chars

; Pattern matching (simple regex alternative)
IF phone?10N { }                     ; exactly 10 digits
IF name?1.A { }                      ; 1+ letters only
    Pattern Meaning
    N Digit 0–9
    A Any letter
    U / L Upper / Lowercase
    E Any character
    1. 1 or more
    n. n or more

    Error Handling¶

    ; Check %Status
SET sc = person.%Save()
IF $$$ISERR(sc) { WRITE $SYSTEM.Status.GetErrorText(sc) }
IF $$$ISOK(sc)  { WRITE "OK" }

; Create error
SET sc = $$$ERROR($$$GeneralError, "Name required")

; TRY/CATCH
TRY {
    SET sc = ..RiskyOperation()
    $$$ThrowOnError(sc)               ; throw if error — clean pattern
} CATCH ex {
    WRITE ex.DisplayString(), !       ; works on ALL exception types
}

; Return error from method
Method Save() As %Status {
    IF ..Name = "" { RETURN $$$ERROR($$$GeneralError, "Name required") }
    RETURN ..%Save()
}
    Macro Meaning
    $$$OK Success status (= 1)
    $$$ISOK(sc) True if OK
    $$$ISERR(sc) True if error
    $$$ERROR(code, msg) Create error status
    $$$ThrowStatus(sc) Throw from status
    $$$ThrowOnError(sc) Throw only if error

    Gotchas

    • DO person.%Save() silently fails — always SET sc = person.%Save()
    • Use ex.DisplayString() in CATCH — works on %Exception.SystemException too
    • Methods should return %Status, not void
